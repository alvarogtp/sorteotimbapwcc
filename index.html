function sortearListaPorDia(jugadoresDia) { let expData = getExposiciones(); // Separación primaria por tipos de cancha let largas = mezclar(jugadoresDia.filter(j => j.tee === 'NEGRA' || j.tee === 'AZUL')); let cortas = mezclar(jugadoresDia.filter(j => j.tee === 'BLANCA')); let grupos = []; function armarGrupos(lista) { while (lista.length >= 3) { // SI QUEDAN 6 JUGADORES: forzar grupo de 3 para que queden 3 y hacer 2 grupos de 3. // SI QUEDAN 5 JUGADORES: forzar grupo de 3 para que queden 2 (que luego se evalúan o fustionan). let tam; if (lista.length === 6 || lista.length === 5) { tam = 3; } else if (lista.length >= 4) { tam = 4; } else { tam = 3; } let grupo = [lista.shift()]; while (grupo.length < tam && lista.length > 0) { let mejorIdx = 0; let menorCosto = Infinity; for (let i = 0; i < lista.length; i++) { let c = evaluarCostoGrupo(grupo, lista[i], expData); if (c < menorCosto) { menorCosto = c; mejorIdx = i; } } grupo.push(lista.splice(mejorIdx, 1)[0]); } grupos.push(grupo); } return lista; // Sobrantes (0, 1 o 2) } let sobLargas = armarGrupos(largas); let sobCortas = armarGrupos(cortas); let sobTotales = [...sobLargas, ...sobCortas]; // Si quedan 3 o más sobrantes entre combinaciones, se crea un grupo mixto de fuerza mayor if (sobTotales.length >= 3) { grupos.push(sobTotales); } else if (sobTotales.length > 0) { // Si sobran 1 o 2, se acoplan a un grupo de 3 existente para hacerlo de 4 sobTotales.forEach(j => { let gIncompleto = grupos.find(g => g.length === 3); if (gIncompleto) { gIncompleto.push(j); } }); } // Registrar exposición a conflictivos en el historial local grupos.forEach(g => { if (g.some(j => j.conducta === 'CONFLICTIVO')) { g.forEach(j => { if (j.conducta !== 'CONFLICTIVO') registrarExposicion(j.id); }); } }); return grupos; }
